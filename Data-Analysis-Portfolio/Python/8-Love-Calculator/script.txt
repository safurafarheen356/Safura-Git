def calculate_love_score(name1, name2):
    name = (name1 + name2).lower()

    true_count = 0
    love_count = 0

    for letter in name:
        if letter in "true":
            true_count += 1
        if letter in "love":
            love_count += 1

    print(f"Love Score = {true_count}{love_count}")

calculate_love_score(
    input("Enter name1: "),
    input("Enter name2: ")
)
